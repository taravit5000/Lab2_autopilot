cmake_minimum_required(VERSION 3.10)
project(lab2_gazebo_capture)
set(CMAKE_CXX_STANDARD 17)
find_package(gazebo REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(lab2_gazebo_capture SHARED ../gazebo_capture.cpp)
target_include_directories(lab2_gazebo_capture PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(lab2_gazebo_capture PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(lab2_gazebo_capture ${GAZEBO_LIBRARIES} gazebo_gui Qt5::Widgets)
